8.SYSTEM.ARRAY CLASS
using System;
using System.Collections.Generic; using System.Text;
namespace ConsoleApplication1
{
class Program
{
static void Main(string[] args)
{
// 1. CREATE ARRAY
Console.WriteLine("1. CREATE ARRAY");


Array a = Array.CreateInstance(typeof(string), 4);


a.SetValue("ram", 0);
a.SetValue("rom", 1);
a.SetValue("cd", 2);
a.SetValue("harddisk", 3);


foreach (string x in a) Console.WriteLine(x);



// 2. ARRAY PROPERTIES
Console.WriteLine("\n2. ARRAY PROPERTIES");


Console.WriteLine("Fixed Size: " + a.IsFixedSize); Console.WriteLine("Length: " + a.Length);

Console.WriteLine("Rank: " + a.Rank);




// 3. LOWER AND UPPER BOUND
Console.WriteLine("\n3. LOWER AND UPPER BOUND");


Array b = Array.CreateInstance(typeof(int), 2, 3, 4);


Console.WriteLine("Lower Bound: " + b.GetLowerBound(1)); Console.WriteLine("Upper Bound: " + b.GetUpperBound(1)); Console.WriteLine("Rank: " + b.Rank);



// 4. SORT
Console.WriteLine("\n4. SORTED ARRAY");


Array.Sort(a);


foreach (string x in a) Console.WriteLine(x);



// 5. BINARY SEARCH
Console.WriteLine("\n5. BINARY SEARCH");


int index = Array.BinarySearch(a, "ram");


if (index >= 0)
Console.WriteLine("Item found at position: " + index);

else
Console.WriteLine("Item not found");




// 6. REVERSE
Console.WriteLine("\n6. REVERSED ARRAY");


Array.Reverse(a);


foreach (string x in a) Console.WriteLine(x);



// 7. CLEAR
Console.WriteLine("\n7. CLEAR ARRAY");


Array.Clear(a, 1, 2);


foreach (string x in a) Console.WriteLine(x);



// 8. COPY
Console.WriteLine("\n8. COPIED ARRAY");


Array target = Array.CreateInstance(typeof(string), 4);


Array.Copy(a, target, 3);

foreach (string x in target) Console.WriteLine(x);

Console.ReadKey();
}
}
}

{
